#include <stdio.h>
#include <unistd.h>
int main() {
    pid_t pid = fork();
    if (pid == 0) {
        printf("This is the child process. PID: %d\n", getpid());
    } else if (pid > 0) {
        printf("This is the parent process. PID: %d\n", getpid());
    } else {
        printf("Fork failed!\n");   }
    return 0;}
Explaination: 
1- when pid==00, it runs the child block and print child pid using getpid() 
2- it runs the parent block and print parent pid using getpid()
3- if pid isn’t =0 or >0, it print else runs and prints fork failed
